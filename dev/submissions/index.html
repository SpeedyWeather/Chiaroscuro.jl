Captain Ahab: Sneaky little whitecaps and a low sun

path: /submissions/whitecaps.jl

rank: 1. of 13 submissions

name = "Captain Ahab"
description = "Sneaky little whitecaps and a low sun"

function albedo(inputs)
    (; land_fraction, sea_ice_concentration, snow_cover) = inputs
    (; cos_zenith, surface_wind_speed, sea_surface_temperature) = inputs

    # water reflects more when the sun is low (small cos_zenith), like a sunset over the sea
    water = 0.05 + 0.15 * (1 - cos_zenith)^5

    # whitecaps from breaking waves cover more of the ocean the stronger the wind
    whitecaps = clamp(3.8e-6 * surface_wind_speed^3.4, 0, 1)
    water = (1 - whitecaps) * water + whitecaps * 0.5

    # melting sea ice (close to freezing) is darker because of melt ponds
    ice = sea_surface_temperature > 272 ? 0.45 : 0.65
    ocean = (1 - sea_ice_concentration) * water + sea_ice_concentration * ice

    land = 0.3 + 0.5 * snow_cover

    a = (1 - land_fraction) * ocean + land_fraction * land
    return clamp(a, 0, 1)
end

Bradley: baseline with lat lon

name = "Bradley"
description = "baseline with lat lon"

function albedo(inputs)
    (; latd, lond, land_fraction, sea_ice_concentration) = inputs
    (; high_vegetation_cover, low_vegetation_cover, low_vegetation_leaf_area) = inputs

    # land: the equation from section 7, with latitude and longitude computed as in section 3
    cos_latd = cosd(latd)
    sin_lond = sind(lond)
    land = 0.9197872 / (((low_vegetation_leaf_area + exp(high_vegetation_cover + cos_latd)) - (sin_lond * (0.28641453 - ((cos_latd * sin_lond) + high_vegetation_cover)))) + low_vegetation_cover)

    # ocean: open water and sea ice (the equation was fitted on land only)
    ocean = 0.06 * (1 - sea_ice_concentration) + 0.6 * sea_ice_concentration

    # weight by the fraction of land in the grid cell, and keep the albedo between 0 and 1
    a = (1 - land_fraction) * ocean + land_fraction * land
    return clamp(a, 0, 1)
end

function albedo(inputs)
    (; land_fraction, sea_ice_concentration, snow_cover) = inputs

    # ocean: blend open ocean and sea ice
    ocean = 0.06 * (1 - sea_ice_concentration) + 0.6 * sea_ice_concentration

    # land: blend bare land and snow
    land = 0.3 * (1 - snow_cover) + 0.8 * snow_cover

    # weight by the fraction of land in the grid cell
    a = (1 - land_fraction) * ocean + land_fraction * land
    return clamp(a, 0, 1)
end

Robin Hood: The forest, dark and deep

name = "Robin Hood"
description = "The forest, dark and deep"

function albedo(inputs)
    (; land_fraction, sea_ice_concentration, snow_depth) = inputs
    (; high_vegetation_cover, low_vegetation_cover, high_vegetation_leaf_area) = inputs

    ocean = 0.06 + 0.54 * sea_ice_concentration

    # bare soil is bright, forests are dark, grass is in between
    bare_soil = max(1 - high_vegetation_cover - low_vegetation_cover, 0)
    land = 0.35 * bare_soil + 0.2 * low_vegetation_cover + 0.13 * high_vegetation_cover

    # snow is brighter the deeper it is, but dense forests (high leaf area) hide it
    snow_cover = snow_depth / (snow_depth + 0.02)
    visible = exp(-0.5 * high_vegetation_leaf_area * high_vegetation_cover)
    land += 0.5 * snow_cover * visible

    a = (1 - land_fraction) * ocean + land_fraction * land
    return clamp(a, 0, 1)
end

    (; high_vegetation_leaf_area, snow_depth, soil_moisture_1) = inputs

    # linear regression on standardised features (training-set mean and std),
    # fitted on all grid cells: land, ocean and sea ice
    sin_lond = sind(lond)
    a = 0.35404542313636833 -
        0.03747681963527709 * (sin_lond - 0.6670778758303375) / 0.274440687488348 -
        0.1581692711679168 * (high_vegetation_cover - 0.23742209374904633) / 0.3569721281528473 -
        0.10786478297918996 * (low_vegetation_cover - 0.4988933205604553) / 0.42057308554649353 -
        0.011140665355197012 * (high_vegetation_leaf_area - 1.2122645378112793) / 1.4638147354125977 -
        0.02054918790951849 * (low_vegetation_leaf_area - 1.0299466848373413) / 0.8743666410446167 +
        0.0473728732784119 * (snow_depth - 0.8201590180397034) / 2.7313692569732666 +
        0.04234441245539077 * (soil_moisture_1 - 0.1947755068540573) / 0.14566275477409363

    return clamp(a, 0, 1)
